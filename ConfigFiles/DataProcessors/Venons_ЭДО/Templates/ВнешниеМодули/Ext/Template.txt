//Модуль кнопки формы:
	Попытка 		
		ВидКонтрагента=Перечисления.ВидКонтрагента.ЮридическоеЛицо;
		ВидКонтрагентаПриИзменении(Неопределено);
		Ответ = Обработки.Venons_ЭДО.Создать().ПолучитьСведенияПоИНН(ИНН);
		ПолноеНаименование=Ответ.fullname;
		Наименование=Ответ.shortName;
		ОКЭД=Ответ.oked;
		КодНДС=Ответ.VATRegCode;
		ЮридическийАдрес=Ответ.address;
		ПочтовыйАдрес=Ответ.address;
		Руководитель=Ответ.director;
		Бухгалтер=Ответ.accountant;
		Если не Ответ.account = "" Тогда
			Счет = Справочники.БанковскиеСчетаКонтрагентов.НайтиПоРеквизиту("НомерСчета",Ответ.account,,Ссылка);
			Если Счет.Пустая() и не Ссылка.Пустая() Тогда
				Счет = Справочники.БанковскиеСчетаКонтрагентов.СоздатьЭлемент();
				Счет.Владелец = Ссылка;
				Счет.Наименование = "Расчетный счет (ГНК)";
				Счет.НомерСчета = Ответ.account;
				Счет.Банк = Справочники.Банки.НайтиПоКоду(Ответ.bankcode);
				Счет.Записать();
				Счет = Счет.Ссылка;
			КонецЕсли;
			ОсновнойБанковскийСчет = Счет;
		КонецЕсли;
	Исключение
	КонецПопытки;

//Конопка отправки ЭСФ через документ 1с
//Модуль кнопки формы:
	Если ЭтоНовый() или ЭтотОбъект.Модифицированность() Тогда
		Предупреждение("Документ не был записан или был изменен. Для отправки его нужно записать!");	
	Иначе
		Обр				= Обработки.Venons_ЭДО.Создать();
		Ответ			= Обр.ВыставитьЭСФDidoxМодульный(Ссылка);
		СостояниеЭСФ	= Ответ.СостояниеЭСФ;
		ИДЭСФ			= Ответ.ИДЭСФ;
		Записать();
	КонецЕсли;

				ЭлементыФормы.ПолеКлючей.СписокВыбора.Добавить(Н,Строчка);
				Н=Н+1;
			КонецЦикла;
